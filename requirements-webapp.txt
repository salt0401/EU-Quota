# Dependencies for the internal quota tracker site (webapp/).
#
# DELIBERATELY SEPARATE from requirements-ci.txt. The daily scrape must not
# grow a web or database dependency: it is the business-critical path, and it
# runs unattended. Nothing in src/ imports webapp/, and the daily task treats
# the ETL step as non-fatal, so the site can be broken or absent entirely
# without affecting the publish.
#
#   <venv>\Scripts\python.exe -m pip install -r requirements-webapp.txt
#
# Major versions capped so an upstream release cannot silently change the site.

Flask>=3.0,<4
SQLAlchemy>=2.0,<3

# Production WSGI server. Flask's built-in server prints "do not use this in a
# production deployment" for good reason: it is single-threaded and has no
# hardening. waitress is the pragmatic Windows choice (no compilation, no
# fork/uwsgi assumptions) and is what serves the site behind the IIS reverse
# proxy:
#
#   <venv>\Scripts\waitress-serve.exe --listen=127.0.0.1:8081 --call webapp.app:create_app
waitress>=3.0,<4

# --- SQL Server ------------------------------------------------------------
# The production store is SQL Server (database MEPSQuota) since 2026-09-27;
# SQLite remains the development default when QUOTA_DB_URL is unset.
#
# Needs the "ODBC Driver 17/18 for SQL Server" on the machine -- verified
# present (Driver 17, 64-bit) on the company server, so no machine-wide
# install is required. See docs/INTERNAL_SITE.md.
pyodbc>=5.0,<6
